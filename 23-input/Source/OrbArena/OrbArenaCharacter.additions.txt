	UPROPERTY(EditAnywhere, Category = "Dash")

		// NEW (lesson 23)
		// BindAction(which action, which moment, which object, which function)
		//   DashAction                    - the Input Action asset (IA_Dash)
		//   ETriggerEvent::Started        - the moment the key goes down
		//                                   (an enum class, lesson 09)
		//   this                          - call the function on THIS character
		//   &AOrbArenaCharacter::Dash     - the address of the member function
		//                                   to call (lessons 11 and 16)
		if (DashAction)
		{
			EnhancedInputComponent->BindAction(DashAction, ETriggerEvent::Started,
				this, &AOrbArenaCharacter::Dash);
		}
		else
		{

		}

// If the template names that variable differently, use the template's name.

void AOrbArenaCharacter::Dash()
{
	UE_LOG(LogTemp, Warning, TEXT("Dash!"));

	// GetActorForwardVector() is a direction of length 1 pointing where the
	// character faces. Times DashStrength, it becomes a velocity.
	//
	// LaunchCharacter(velocity, replace XY speed?, replace Z speed?)
	//   true  - replace the horizontal speed (a sharp dash, not a nudge)
	//   false - keep the vertical speed (so gravity still works)
	LaunchCharacter(GetActorForwardVector() * DashStrength, true, false);
}
