	/** NEW (milestone 1): enemies closer than this when you dash get stunned. */
	UPROPERTY(EditAnywhere, Category = "Dash")
	float StunRadius = 300.f;

	/** NEW (milestone 1): how long a stun lasts, in seconds. */
	UPROPERTY(EditAnywhere, Category = "Dash")
	float StunSeconds = 2.f;

// ----------------------------------------------------------------------------
// 1B. OrbArenaCharacter.cpp, includes at the top (below the existing ones)
// ----------------------------------------------------------------------------

#include "Enemy.h"                     // NEW (milestone 1): AEnemy, Stun
#include "Kismet/GameplayStatics.h"    // NEW (milestone 1): GetAllActorsOfClass

// ----------------------------------------------------------------------------
// 1C. OrbArenaCharacter.cpp, REPLACE lesson 23's Dash() with this one
// ----------------------------------------------------------------------------

void AOrbArenaCharacter::Dash()
{
	LaunchCharacter(GetActorForwardVector() * DashStrength, true, false);

	// NEW (milestone 1): stun every enemy close enough.
	// GetAllActorsOfClass, lesson 26.
	TArray<AActor*> Enemies;
	UGameplayStatics::GetAllActorsOfClass(this, AEnemy::StaticClass(), Enemies);
	for (AActor* Actor : Enemies)
	{
		AEnemy* Enemy = Cast<AEnemy>(Actor);
		if (Enemy != nullptr
			&& FVector::Dist(GetActorLocation(), Enemy->GetActorLocation()) < StunRadius)
		{
			Enemy->Stun(StunSeconds);
		}
	}
}

void AOrbArenaCharacter::RestartGame()
{
	// The GameMode decides whether restarting is allowed right now.
	if (AOrbArenaGameMode* GameMode = GetWorld()->GetAuthGameMode<AOrbArenaGameMode>())
	{
		GameMode->RestartRound();
	}
}
